# Linguitect to Rust Adapter Specification

## 1. Adapter Metadata

```
@LANGUAGE_ADAPTER "Linguitect_to_Rust"
  @VERSION "1.0"
  @LANGUAGE_VERSION "Rust 1.54+"
  @TYPE "export"
  
  @METADATA
    @PARADIGMS ["procedural", "object_oriented", "functional"]
    @TYPING type="static" strength="strong"
    @MEMORY_MANAGEMENT type="ownership"
    @EXECUTION_MODEL "compiled"
    @TARGET_RUST_VERSION "1.58" # Default target version
  @END
```

## 2. Code Generation Rules

### 2.1 Program and Module Structure

```
@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PROGRAM {name}
      @MODULE {module_name}
        {module_content}
      @ENDMODULE
    @END
  @END
  
  @TARGET_CODE
    // {name}
    // Generated by Linguitect-to-Rust Adapter v1.0
    
    {transformed_module_content}
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @MODULE {name}
      {module_content}
    @ENDMODULE
  @END
  
  @TARGET_CODE
    mod {name} {
        {transformed_module_content}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @MODULE {name}
      @NOTE "This module is public"
      {module_content}
    @ENDMODULE
  @END
  
  @TARGET_CODE
    pub mod {name} {
        {transformed_module_content}
    }
  @END
@END
```

  @END
@END

  @END

  @END
  
  @VARIANTS

  @END

### 2.7 Control Flow

```
@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @IF {condition}
      {true_body}
    @ELSE
      {false_body}
    @ENDIF
  @END
  
  @TARGET_CODE
    if {transformed_condition} {
        {transformed_true_body}
    } else {
        {transformed_false_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @IF {condition}
      {true_body}
    @ELIF {elif_condition}
      {elif_body}
    @ELSE
      {else_body}
    @ENDIF
  @END
  
  @TARGET_CODE
    if {transformed_condition} {
        {transformed_true_body}
    } else if {transformed_elif_condition} {
        {transformed_elif_body}
    } else {
        {transformed_else_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @MATCH {expression}
      @CASE {pattern}
        {body}
      @DEFAULT
        # No operation
    @ENDMATCH
  @END
  
  @TARGET_CODE
    if let {transformed_pattern} = {transformed_expression} {
        {transformed_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PATTERN_MATCH {expression}
      @PATTERN {pattern1}
        {body1}
      @PATTERN {pattern2}
        {body2}
      @DEFAULT
        {default_body}
    @ENDPATTERN_MATCH
  @END
  
  @TARGET_CODE
    match {transformed_expression} {
        {transformed_pattern1} => {
            {transformed_body1}
        },
        {transformed_pattern2} => {
            {transformed_body2}
        },
        _ => {
            {transformed_default_body}
        },
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PATTERN_MATCH {expression}
      @PATTERN {pattern1} IF {guard1}
        {body1}
      @PATTERN {pattern2}
        {body2}
      @DEFAULT
        {default_body}
    @ENDPATTERN_MATCH
  @END
  
  @TARGET_CODE
    match {transformed_expression} {
        {transformed_pattern1} if {transformed_guard1} => {
            {transformed_body1}
        },
        {transformed_pattern2} => {
            {transformed_body2}
        },
        _ => {
            {transformed_default_body}
        },
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="for" var={var} in={iterable}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    for {var} in {transformed_iterable} {
        {transformed_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="for-range" var={var} from={start} to={end}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    for {var} in {transformed_start}..{transformed_end} {
        {transformed_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="for-range" var={var} from={start} to={end} step={step}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    for {var} in ({transformed_start}..{transformed_end}).step_by({transformed_step} as usize) {
        {transformed_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="while" {condition}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    while {transformed_condition} {
        {transformed_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="infinite"
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    loop {
        {transformed_body}
    }
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="while-pattern" pattern={pattern} expression={expression}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE

    @ENDLOOP
  @END

  @END
@END

@CODE_GEN_RULE

  @END

@END

@CODE_GEN_RULE
