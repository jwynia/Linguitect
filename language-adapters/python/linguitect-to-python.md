# Linguitect to Python Adapter Specification

## 1. Adapter Metadata

```
@LANGUAGE_ADAPTER "Linguitect_to_Python"
  @VERSION "1.0"
  @LANGUAGE_VERSION "Python 3.6-3.11"
  @TYPE "export"
  
  @METADATA
    @PARADIGMS ["procedural", "object_oriented", "functional"]
    @TYPING type="dynamic" strength="strong"
    @MEMORY_MANAGEMENT type="gc"
    @EXECUTION_MODEL "interpreted"
    @TARGET_PYTHON_VERSION "3.9" # Default target version
  @END
```

## 2. Code Generation Rules

### 2.1 Program and Module Structure

```
@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PROGRAM {name}
      @MODULE {module_name}
        {module_content}
      @ENDMODULE
    @END
  @END
  
  @TARGET_CODE
    # {name}
    # Generated by Linguitect-to-Python Adapter v1.0
    
    {transformed_module_content}
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @MODULE {name} [FROM {parent}]
      {module_content}
    @ENDMODULE
  @END
  
  @TARGET_CODE
    # Module: {name}
    {transformed_module_content}
  @END
@END
```

### 2.5 Class Definitions

```
@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @CLASS {name} [EXTENDS {parent_classes}] [IMPLEMENTS {interfaces}]
      @DOCS {docstring}
      {body}
    @ENDCLASS
  @END
  
  @TARGET_CODE
    class {name}{parenthesized_inheritance(parent_classes, interfaces)}:
        """{docstring}"""
        {transformed_body}
  @END
  
  @CONTEXT
    @FUNCTION parenthesized_inheritance(parents, interfaces)
      parents_list = []
      if parents:
          parents_list.extend(parents.split(', '))
      if interfaces:
          # Python doesn't have interfaces, so we treat them as mixins
          parents_list.extend(interfaces.split(', '))
      
      if parents_list:
          return "(" + ", ".join(parents_list) + ")"
      else:
          return ""
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @CONSTRUCTOR({params})
      @BODY
        {body}
      @END
    @ENDCONSTRUCTOR
  @END
  
  @TARGET_CODE
    def __init__(self, {transform_params_to_python(params)}):
        {transformed_body}
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.6 && add_type_hints"
      def __init__(self, {transform_params_with_type_hints(params)}) -> None:
          {transformed_body}
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @METHOD {name}({params}) -> {return_type} [access={access_level}] [static={is_static}]
      @DOCS {docstring}
      @BODY
        {body}
      @END
    @ENDMETHOD
  @END
  
  @TARGET_CODE
    {staticmethod_decorator if is_static else ""}
    def {name}({method_params_transform(params, is_static)}):
        """{docstring}"""
        {transformed_body}
  @END
  
  @CONTEXT
    @FUNCTION staticmethod_decorator
      return "@staticmethod\n"
    @END
    
    @FUNCTION method_params_transform(params, is_static)
      if is_static:
          return transform_params_to_python(params)
      else:
          return "self" + (", " + transform_params_to_python(params) if params else "")
    @END
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.6 && add_type_hints"
      {staticmethod_decorator if is_static else ""}
      def {name}({method_params_with_type_hints(params, is_static)}) -> {type_hint_transform(return_type)}:
          """{docstring}"""
          {transformed_body}
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PROPERTY {name} {type} [access={access_level}] [static={is_static}] [= {default_value}]
  @END
  
  @TARGET_CODE
    {name_with_access_convention(name, access_level)} = {transformed_default_value if has_default else "None"}
  @END
  
  @CONTEXT
    @FUNCTION name_with_access_convention(name, access)
      if access == "private" or access == "protected":
          return "_" + name
      else:
          return name
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @METHOD {name}({params}) -> {return_type}
      @DECORATOR property
      @BODY
        {body}
      @END
    @ENDMETHOD
  @END
  
  @TARGET_CODE
    @property
    def {name}({method_params_transform(params, false)}):
        {transformed_body}
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.6 && add_type_hints"
      @property
      def {name}(self) -> {type_hint_transform(return_type)}:
          {transformed_body}
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PROPERTY_GET {object}.{property}
  @END
  
  @TARGET_CODE
    {transformed_object}.{property}
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PROPERTY_SET {object}.{property} <- {value}
  @END
  
  @TARGET_CODE
    {transformed_object}.{property} = {transformed_value}
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @THIS
  @END
  
  @TARGET_CODE
    self
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @SUPER
  @END
  
  @TARGET_CODE
    super()
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @NEW {class_name} WITH ({args}) -> {var}
  @END
  
  @TARGET_CODE
    {var} = {class_name}({transformed_args})
  @END
@END
```

### 2.6 Type Definitions

```
@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @STRUCT {name}
      @FIELD {field1} {type1} [default={value1}]
      @FIELD {field2} {type2} [default={value2}]
      ...
    @ENDSTRUCT
  @END
  
  @TARGET_CODE
    from collections import namedtuple
    
    {name} = namedtuple('{name}', [{field_list}])
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.7"
      from dataclasses import dataclass
      from typing import Optional
      
      @dataclass
      class {name}:
          {typed_fields_with_defaults}
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @ENUM {name} [type={base_type}]
      @VALUE {value1} [= {explicit_value1}]
      @VALUE {value2} [= {explicit_value2}]
      ...
    @ENDENUM
  @END
  
  @TARGET_CODE
    class {name}:
        {enum_value_assignments}
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.4"
      from enum import Enum
      
      class {name}(Enum):
          {enum_value_assignments}
    @END
    
    @VARIANT when="python_version >= 3.6 && base_type"
      from enum import IntEnum, StrEnum  # StrEnum in 3.11+
      
      class {name}({get_enum_base_type(base_type)}):
          {enum_value_assignments}
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @TYPE_ALIAS {name} = {existing_type}
  @END
  
  @TARGET_CODE
    # Type aliases require type annotations
    # This will be a comment in Python versions < 3.6
    {name} = {existing_type}
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.6"
      from typing import TypeAlias
      
      {name}: TypeAlias = {type_hint_transform(existing_type)}
    @END
  @END
@END
```

```
@END

    @ENDLOOP
  @END

  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="while" {condition}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    while {transformed_condition}:
        {indent(transformed_body)}
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @LOOP type="do-while" {condition}
      {body}
    @ENDLOOP
  @END
  
  @TARGET_CODE
    # Python does not have do-while loops
    # Using a while True with a conditional break
    while True:
        {indent(transformed_body)}
        if not ({transformed_condition}):
            break
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @BREAK [label]
  @END
  
  @TARGET_CODE
    break  # Python does not support labeled breaks
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @CONTINUE [label]
  @END
  
  @TARGET_CODE
    continue  # Python does not support labeled continues
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @SWITCH {expression}
      @CASE {value1}
        {statements1}
      @CASE {value2}
        {statements2}
      @DEFAULT
        {default_statements}
    @ENDSWITCH
  @END
  
  @TARGET_CODE
    # Python 3.10+ has match statements, but for compatibility using if-elif
    {expr_var} = {transformed_expression}
    if {expr_var} == {transformed_value1}:
        {indent(transformed_statements1)}
    elif {expr_var} == {transformed_value2}:
        {indent(transformed_statements2)}
    else:
        {indent(transformed_default_statements)}
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.10"
      match {transformed_expression}:
          case {transformed_value1}:
              {indent(transformed_statements1)}
          case {transformed_value2}:
              {indent(transformed_statements2)}
          case _:
              {indent(transformed_default_statements)}
    @END
  @END
@END

@CODE_GEN_RULE
  @LINGUITECT_PATTERN
    @PATTERN_MATCH {expression}
      @PATTERN {pattern1} [IF {condition1}]
        {statements1}
      @PATTERN {pattern2} [IF {condition2}]
        {statements2}
      @DEFAULT
        {default_statements}
    @ENDPATTERN_MATCH
  @END
  
  @TARGET_CODE
    # Pattern matching in Python < 3.10 requires manual unpacking and conditions
    {impl_legacy_pattern_matching(expression, patterns, conditions, statements)}
  @END
  
  @VARIANTS
    @VARIANT when="python_version >= 3.10"
      match {transformed_expression}:
          case {transformed_pattern1} if {transformed_condition1}:
              {indent(transformed_statements1)}
          case {transformed_pattern2} if {transformed_condition2}:
              {indent(transformed_statements2)}
          case _:
              {indent(transformed_default_statements)}
    @END
  @END
@END
```

## 3. Type Mapping

### 3.1 Linguitect to Python Type Mapping

```
@TYPE_MAPPING
  @LINGUITECT_TYPE
```

## 6. Common Transformation Functions

```
@TRANSFORMATION_FUNCTIONS
  @FUNCTION indent(code, spaces=4)
    lines = code.split('\n')
    return '\n'.join(' ' * spaces + line for line in lines)
  @END
  
  @FUNCTION transform_type_annotation(type)
    # Maps Linguitect type annotations to Python type annotations
    if type.startswith('@INT'):
      return 'int'
    elif type.startswith('@FLOAT'):
      return 'float'
    elif type.startswith('@STRING'):
      return 'str'
    elif type.startswith('@BOOLEAN'):
      return 'bool'
    elif type.startswith('@LIST'):
      element_type = extract_element_type(type)
      return f'List[{transform_type_annotation(element_type)}]'
    # ... other type transformations
  @END
  
  @FUNCTION transform_params_to_python(params)
    # Convert Linguitect params to Python param syntax
    result = []
    for param in params.split(','):
      name_type = param.split(':')
      name = name_type[0].strip()
      # Handle default values
      if '=' in name:
        name_parts = name.split('=')
        result.append(f"{name_parts[0].strip()} = {name_parts[1].strip()}")
      else:
        result.append(name)
    return ', '.join(result)
  @END
  
  @FUNCTION transform_params_with_type_hints(params)
    # Convert Linguitect params to Python params with type hints
    result = []
    for param in params.split(','):
      if ':' in param:
        name_type = param.split(':')
        name = name_type[0].strip()
        type_hint = transform_type_annotation(name_type[1].strip())
        
        # Handle default values
        if '=' in name:
          name_parts = name.split('=')
          result.append(f"{name_parts[0].strip()}: {type_hint} = {name_parts[1].strip()}")
        else:
          result.append(f"{name}: {type_hint}")
      else:
        result.append(param.strip())
    return ', '.join(result)
  @END
  
  @FUNCTION transform_lambda_body(body)
    # Extract the return expression from a lambda body
    return_stmt = extract_return_statement(body)
    if return_stmt:
      return transform_expression(return_stmt.expression)
    return "None"
  @END
  
  @FUNCTION method_params_with_type_hints(params, is_static)
    if is_static:
      return transform_params_with_type_hints(params)
    else:
      # Add self parameter
      if params:
        return f"self, {transform_params_with_type_hints(params)}"
      else:
        return "self"
  @END
  
  @FUNCTION python_type_constructor(type)
    # Get Python constructor for a Linguitect type
    type_map = {
      '@INT': 'int',
      '@FLOAT': 'float',
      '@STRING': 'str',
      '@BOOLEAN': 'bool',
      '@LIST': 'list',
      '@MAP': 'dict',
      '@SET': 'set',
      '@TUPLE': 'tuple'
    }
    
    for prefix, constructor in type_map.items():
      if type.startswith(prefix):
        return constructor
    
    return 'object'  # fallback
  @END
@END
```
